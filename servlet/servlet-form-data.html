Servlet: solicitudes HTTP del cliente

Servlet Datos de formulario

En muchos casos, es necesario pasar información desde el navegador al servidor web y, finalmente, al programa de backend. El navegador utiliza dos métodos para pasar esta información al servidor web: el método GET y el método POST.

Método GET

El método GET envía información de usuario codificada a la solicitud de página. La página y la información codificada están separadas por el carácter ?, como se muestra a continuación:

http://www.test.com/hello?key1=value1&key2=value2

El método GET es el método predeterminado para pasar información del navegador al servidor web. Produce una cadena larga que aparece en la barra de direcciones del navegador. Si va a pasar contraseñas u otra información sensible al servidor, no utilice el método GET. El método GET tiene una limitación de tamaño: la cadena de solicitud puede contener como máximo 1024 caracteres.

Esta información se transmite mediante el encabezado QUERY_STRING y se puede acceder a ella a través de la variable de entorno QUERY_STRING. Servlet usadoGet()el método para procesar este tipo de solicitudes.

Método POST

Otro método relativamente fiable para pasar información al programa de backend es el método POST. El método POST empaqueta la información de manera casi igual que el método GET, pero en lugar de enviar la información como una cadena de texto después del carácter ? en la URL, la envía como un mensaje separado. El mensaje se transmite al programa de backend como salida estándar, y usted puede analizar y utilizar esa salida estándar. Servlet usa el método doPost() para procesar este tipo de solicitudes.

Usar Servlet para leer datos de formulario

Servlet procesa los datos de formulario, que se analizan automáticamente mediante diferentes métodos según la situación:

  • getParameter()：Puede llamar al método request.getParameter() para obtener el valor de un parámetro de formulario.
  • getParameterValues()：Si el parámetro aparece más de una vez, llame a este método y devolverá múltiples valores, por ejemplo en casillas de verificación.
  • getParameterNames()：Si desea obtener una lista completa de todos los parámetros de la solicitud actual, llame a este método.

Ejemplo del método GET usando URL

A continuación se muestra una URL simple que usará el método GET para pasar dos valores al programa HelloForm.

http://localhost:8080/TomcatTest/HelloForm?name=菜鸟教程&url=www.runoob.com

El siguiente es un programa que procesa la entrada del navegador webHelloForm.javaServlet. Usaremos elgetParameter()método, con el que podemos acceder fácilmente a la información transmitida:

package com.runoob.test;

import java.io.IOException;
import java.io.PrintWriter;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * Servlet implementation class HelloForm
 */
@WebServlet("/HelloForm")
public class HelloForm extends HttpServlet {
    private static final long serialVersionUID = 1L;
       
    /**
     * @see HttpServlet#HttpServlet()
     */
    public HelloForm() {
        super();
        // TODO Auto-generated constructor stub
    }

    /**
     * @see HttpServlet#doGet(HttpServletRequest request, HttpServletResponse response)
     */
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        // 设置响应内容类型
        response.setContentType("text/html;charset=UTF-8");

        PrintWriter out = response.getWriter();
        String title = "使用 GET 方法读取表单数据";
        // 处理中文
        String name =new String(request.getParameter("name").getBytes("ISO-8859-1"),"UTF-8");
        String docType = "<!DOCTYPE html> \n";
        out.println(docType +
            "<html>\n" +
            "<head><title>" + title + "</title></head>\n" +
            "<body bgcolor=\"#f0f0f0\">\n" +
            "<h1 align=\"center\">" + title + "</h1>\n" +
            "<ul>\n" +
            "  <li><b>站点名</b>："
            + name + "\n" +
            "  <li><b>网址</b>："
            + request.getParameter("url") + "\n" +
            "</ul>\n" +
            "</body></html>");
    }
    
    // 处理 POST 方法请求的方法
    public void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        doGet(request, response);
    }
}

package com.runoob.test;

import java.io.IOException;
import java.io.PrintWriter;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * Servlet implementation class CheckBox
 */
@WebServlet("/CheckBox")
public class CheckBox extends HttpServlet {
    private static final long serialVersionUID = 1L;
    
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        
        // 设置响应内容类型
        response.setContentType("text/html;charset=UTF-8");

        PrintWriter out = response.getWriter();
        String title = "读取复选框数据";
        String docType = "<!DOCTYPE html> \n";
            out.println(docType +
                "<html>\n" +
                "<head><title>" + title + "</title></head>\n" +
                "<body bgcolor=\"#f0f0f0\">\n" +
                "<h1 align=\"center\">" + title + "</h1>\n" +
                "<ul>\n" +
                "  <li><b>菜鸟按教程标识：</b>: "
                + request.getParameter("runoob") + "\n" +
                "  <li><b>Google 标识：</b>: "
                + request.getParameter("google") + "\n" +
                "  <li><b>淘宝标识：</b>: "
                + request.getParameter("taobao") + "\n" +
                "</ul>\n" +
                "</body></html>");
    }
    
    // 处理 POST 方法请求的方法
    public void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        doGet(request, response);
    }
}

Leer todos los parámetros del formulario

El siguiente es un ejemplo genérico que utiliza elgetParameterNames()método de HttpServletRequest para leer todos los parámetros de formulario disponibles. Este método devuelve una enumeración que contiene los nombres de los parámetros en un orden no especificado.

Una vez que tenemos una enumeración, podemos recorrerla de manera estándar usando elhasMoreElements()método para determinar cuándo detenernos, usando elnextElement()método para obtener el nombre de cada parámetro.

import java.io.IOException;
import java.io.PrintWriter;
import java.util.Enumeration;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * Servlet implementation class ReadParams
 */
@WebServlet("/ReadParams")
public class ReadParams extends HttpServlet {
    private static final long serialVersionUID = 1L;
       
    /**
     * @see HttpServlet#HttpServlet()
     */
    public ReadParams() {
        super();
        // TODO Auto-generated constructor stub
    }

    /**
     * @see HttpServlet#doGet(HttpServletRequest request, HttpServletResponse response)
     */
    protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        // 设置响应内容类型
        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();
        String title = "读取所有的表单数据";
        String docType =
            "<!doctype html public \"-//w3c//dtd html 4.0 " +
            "transitional//en\">\n";
            out.println(docType +
            "<html>\n" +
            "<head><meta charset=\"utf-8\"><title>" + title + "</title></head>\n" +
            "<body bgcolor=\"#f0f0f0\">\n" +
            "<h1 align=\"center\">" + title + "</h1>\n" +
            "<table width=\"100%\" border=\"1\" align=\"center\">\n" +
            "<tr bgcolor=\"#949494\">\n" +
            "<th>参数名称</th><th>参数值</th>\n"+
            "</tr>\n");

        Enumeration paramNames = request.getParameterNames();

        while(paramNames.hasMoreElements()) {
            String paramName = (String)paramNames.nextElement();
            out.print("<tr><td>" + paramName + "</td>\n");
            String[] paramValues =
            request.getParameterValues(paramName);
            // 读取单个值的数据
            if (paramValues.length == 1) {
                String paramValue = paramValues[0];
                if (paramValue.length() == 0)
                    out.println("<td><i>没有值</i></td>");
                else
                    out.println("<td>" + paramValue + "</td>");
            } else {
                // 读取多个值的数据
                out.println("<td><ul>");
                for(int i=0; i < paramValues.length; i++) {
                out.println("<li>" + paramValues[i]);
            }
                out.println("</ul></td>");
            }
            out.print("</tr>");
        }
        out.println("\n</table>\n</body></html>");
    }

    /**
     * @see HttpServlet#doPost(HttpServletRequest request, HttpServletResponse response)
     */
    protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException {
        // TODO Auto-generated method stub
        doGet(request, response);
    }

}

    Si utiliza getparameter() para obtener caracteres chinos, solo aparecen signos de interrogación:

    String name =new String(request.getParameter("name").getBytes("ISO8859-1"),"UTF-8");

    Debido a que la codificación predeterminada de tomcat8 es UTF-8, y este filtro lo decodifica como ISO8859-1 (es decir, lo decodifica dos veces), se producirá un problema que hará que todos los caracteres chinos se conviertan en signos de interrogación.

    Por lo tanto, solo necesitamos no usar este filtro.

    Es decir, adoptar el siguiente código:

    String name =new String(request.getParameter("name"));

    Así los caracteres chinos se mostrarán correctamente.
