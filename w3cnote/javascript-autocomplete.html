JavaScript implementa la función de autocompletar en el cuadro de búsqueda (autocomplete)

Categoría Tecnología de programación

En este capítulo aprenderemos a usar JS/CSS para implementar la función de autocompletar en el cuadro de búsqueda.

Primero veamos el efecto a continuación:

Ver el ejemplo en línea:https://c.example.com/codedemo/6190

Código HTML básico

Ejemplo

<!--autocomplete="off" asegura que el formulario tenga desactivada la función de autocompletar:--> <form autocomplete="off" action="/index.php"> <div class="autocomplete" style="width:300px;"> <input id="myInput" type="text" name="s" placeholder="Introduce el contenido de búsqueda"> </div> <input type="submit"> </form>

Luego preparamos los datos de prueba básicos de búsqueda, que son un array de JavaScript:

var sites = ["Google","Taobao","Ejemplo","Wiki","Zhihu","Baidu","Sina","Tmall","JD","Alibaba","QQ","Weixin"];

A continuación, los estilos del cuadro de búsqueda y del menú de sugerencias:

Ejemplo

* { box-sizing: border-box; } body { font: 16px Arial; } .autocomplete { /*La posición del contenedor se establece como relative:*/ position: relative; display: inline-block; } input { border: 1px solid transparent; background-color: #f1f1f1; padding: 10px; font-size: 16px; } input[type=text] { background-color: #f1f1f1; width: 100%; } input[type=submit] { background-color: DodgerBlue; color: #fff; } .autocomplete-items { position: absolute; border: 1px solid #d4d4d4; border-bottom: none; border-top: none; z-index: 99; /*Configurar el ancho de los elementos de autocompletar igual al del contenedor.*/ top: 100%; left: 0; right: 0; } .autocomplete-items div { padding: 10px; cursor: pointer; background-color: #fff; border-bottom: 1px solid #d4d4d4; } .autocomplete-items div:hover { /*El color de fondo establecido al pasar el mouse sobre un elemento de autocompletar.*/ background-color: #e9e9e9; } .autocomplete-active { /*El color de fondo al navegar por los elementos con las teclas de flecha.*/ background-color: DodgerBlue !important; color: #ffffff; }

A continuación se muestra el código JavaScript del cuadro de búsqueda y el menú de sugerencias:

Ejemplo

function autocomplete(inp, arr) { /*La función de autocompletar tiene dos parámetros: el elemento del cuadro de entrada (input) y el array de autocompletar.*/ var currentFocus; /*Escuchar el cuadro de entrada (input); cuando se esté en el elemento del cuadro de entrada, ejecutar la siguiente función.*/ inp.addEventListener("input", function(e) { var a, b, i, val = this.value; /*Cerrar la lista de autocompletar ya abierta.*/ closeAllLists(); if (!val) { return false;} currentFocus = -1; /*Crear un elemento DIV para colocar los valores de la lista de autocompletar.*/ a = document.createElement("DIV"); a.setAttribute("id", this.id + "autocomplete-list"); a.setAttribute("class", "autocomplete-items"); /*El DIV como elemento hijo del contenedor de autocompletar.*/ this.parentNode.appendChild(a); /*Iterar sobre el array.*/ for (i = 0; i < arr.length; i++) { /*Comprobar si algún elemento de autocompletar tiene el mismo contenido que el valor del campo de texto, sin distinguir mayúsculas de minúsculas.*/ if (arr[i].substr(0, val.length).toUpperCase() == val.toUpperCase()) { /*Crear un elemento DIV para cada elemento coincidente.*/ b = document.createElement("DIV"); /*Resaltar en negrita el elemento coincidente.*/ b.innerHTML = "<strong>" + arr[i].substr(0, val.length) + "</strong>"; b.innerHTML += arr[i].substr(val.length); /*El elemento seleccionado se inserta en un campo de entrada (input) oculto, para guardar el valor actualmente seleccionado.*/ b.innerHTML += "<input type='hidden' value='" + arr[i] + "'>"; /*Cuando alguien hace clic en un elemento de autocompletar (elemento DIV), se ejecuta la función.*/ b.addEventListener("click", function(e) { /*El elemento seleccionado se inserta en el campo de búsqueda (input) oculto.*/ inp.value = this.getElementsByTagName("input")[0].value; /*Cerrar la lista de autocompletar.*/ closeAllLists(); }); a.appendChild(b); } } }); /*Ejecutar la función cuando se presiona una tecla en el teclado.*/ inp.addEventListener("keydown", function(e) { var x = document.getElementById(this.id + "autocomplete-list"); if (x) x = x.getElementsByTagName("div"); if (e.keyCode == 40) { /*Si se presiona la tecla de flecha hacia abajo, la variable currentFocus se incrementa en 1, es decir, se mueve una posición hacia abajo.*/ currentFocus++; /*Hacer que el elemento actualmente seleccionado sea más visible.*/ addActive(x); } else if (e.keyCode == 38) { //up /*Al presionar la tecla de flecha hacia arriba, el elemento seleccionado de la lista se mueve una posición hacia arriba.*/ currentFocus--; /*Hacer que el elemento actualmente seleccionado sea más visible.*/ addActive(x); } else if (e.keyCode == 13) { /*Si se presiona la tecla ENTER, se evita el envío; también puedes configurar el envío con submit.*/ e.preventDefault(); if (currentFocus > -1) { /*Simular un clic en el elemento seleccionado.*/ if (x) x[currentFocus].click(); } } }); function addActive(x) { /*Función para establecer la opción seleccionada.*/ if (!x) return false; /*Al mover la opción, se establece el color de fondo de la opción seleccionada actual.*/ removeActive(x); if (currentFocus >= x.length) currentFocus = 0; if (currentFocus < 0) currentFocus = (x.length - 1); /*Añadir la clase "autocomplete-active".*/ x[currentFocus].classList.add("autocomplete-active"); } function removeActive(x) { /*Eliminar la clase "autocomplete-active" de las opciones no seleccionadas.*/ for (var i = 0; i < x.length; i++) { x[i].classList.remove("autocomplete-active"); } } function closeAllLists(elmnt) { /*Cerrar la lista de autocompletar.*/ var x = document.getElementsByClassName("autocomplete-items"); for (var i = 0; i < x.length; i++) { if (elmnt != x[i] && elmnt != inp) { x[i].parentNode.removeChild(x[i]); } } } /*Hacer clic en cualquier parte del documento HTML para cerrar la lista de autocompletar.*/ document.addEventListener("click", function (e) { closeAllLists(e.target); }); }

Finalmente, usamos el código JavaScript anterior en el cuadro de entrada con id "myInput":

Ejemplo

autocomplete(document.getElementById("myInput"), countries);
