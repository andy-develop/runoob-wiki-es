Eliminar elementos de un array en JavaScript

Categoría Programación

Este artículo explica cómo eliminar elementos de un array en JavaScript.

En el siguiente ejemplo, definimos un método personalizado para eliminar elementos de un array:

Ejemplo

Array.prototype.removeByValue = function (val) {
  for (var i = 0; i < this.length; i++) {
    if (this[i] === val) {
      this.splice(i, 1);
      i--;
    }
  }
  return this;
}

var sites = ['apple', 'google', 'ejemplo', 'taobao'];
sites.removeByValue('google');

console.log(sites);
// -> ['apple', 'ejemplo', 'taobao']

const array = [2, 5, 9];

console.log(array);

const index = array.indexOf(5);
if (index > -1) {
  array.splice(index, 1); // El segundo parámetro es el número de eliminaciones; se configura para eliminar solo una vez
}

// array = [2, 9]

El siguiente ejemplo muestra cómo eliminar uno o más elementos de un array:

Ejemplo

function removeItemOnce(arr, value) {
  var index = arr.indexOf(value);
  if (index > -1) {
    arr.splice(index, 1);
  }
  return arr;
}

function removeItemAll(arr, value) {
  var i = 0;
  while (i < arr.length) {
    if (arr[i] === value) {
      arr.splice(i, 1);
    } else {
      ++i;
    }
  }
  return arr;
}
// Usage
console.log(removeItemOnce([2,5,9,1,5,8,5], 5))
console.log(removeItemAll([2,5,9,1,5,8,5], 5))
