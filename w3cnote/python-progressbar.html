Barra de progreso de Python

Categoría Tecnología de programación

En Python hay varias formas de implementar barras de progreso; a continuación se presentan algunos métodos comunes:

Ejemplo

import time
for i in range(101): # Agregar gráfico de barra de progreso y porcentaje
    bar = '[' + '=' * (i // 2) + ' ' * (50 - i // 2) + ']'
    print(f"\r{bar} {i:3}%", end='', flush=True)
    time.sleep(0.05)
print()

Ejemplo

import sys
import time

def progress_bar(current, total, bar_length=50):
    percent = float(current) / total
    arrow = '=' * int(round(percent * bar_length) - 1) + '>'
    spaces = ' ' * (bar_length - len(arrow))
   
    sys.stdout.write(f"\rProgreso: [{arrow + spaces}] {int(round(percent * 100))}%")
    sys.stdout.flush()

for i in range(101):
    time.sleep(0.1)
    progress_bar(i, 100)
print()  # Salto de línea

Usar diferentes colores para el progreso:

Ejemplo

import time
import sys

def progress_bar(current, total, bar_length=50):
    """
Mostrar barra de progreso
:param current: progreso actual
:param total: progreso total
:param bar_length: longitud de la barra de progreso
    """
    percent = current / total
    arrow = '=' * int(round(percent * bar_length) - 1) + '>'
    spaces = ' ' * (bar_length - len(arrow))
   
    # Agregar color (opcional)
    color_code = 32  # Verde
    if percent > 0.7:
        color_code = 33  # Amarillo
    if percent > 0.9:
        color_code = 31  # Rojo
   
    # Formatear salida
    sys.stdout.write(f"\r\033[{color_code}mProgreso: [{arrow + spaces}] {current:3d}/{total} ({percent:.0%})\033[0m")
    sys.stdout.flush()

# Ejemplo de uso
total = 100
for i in range(total + 1):
    progress_bar(i, total)
    time.sleep(0.05)

print("\n¡Completado!")  # Salto de línea al completar

Usando la biblioteca estándar tqdm

tqdm es una de las bibliotecas de barras de progreso más populares, sencilla de instalar y fácil de usar.

Instalación de la biblioteca estándar con pip:

pip install tqdm

Ejemplo

from tqdm import tqdm
import time

# Uso básico
for i in tqdm(range(100)):
    time.sleep(0.1)  # Simular una tarea

# Barra de progreso con descripción
with tqdm(range(100), desc="Progreso del proceso") as pbar:
    for i in pbar:
        time.sleep(0.1)
       
# Actualización manual
pbar = tqdm(total=100)
for i in range(10):
    time.sleep(0.5)
    pbar.update(10)  # Actualizar 10 cada vez
pbar.close()
